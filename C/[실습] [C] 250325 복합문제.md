```c
// connection.c

#include "connection.h"

OCIEnv* envhp;
OCIError* errhp;
OCISvcCtx* svchp;
OCISession* usrhp;
OCIServer* srvhp;
OCIStmt* stmthp;
OCIParam* param;

OCIDefine* def1 = NULL, * def2 = NULL, * def3 = NULL, * def4 = NULL, * def5 = NULL, * def6 = NULL;
OCIBind* bnd1 = NULL, * bnd2 = NULL, * bnd3 = NULL, * bnd4 = NULL, * bnd5 = NULL;

sword status;

void connect() {
    char* username = "C##DEV";
    char* password = "1234";
    char* dbname = "localhost:1521/xe";

    OCIEnvCreate(&envhp, OCI_DEFAULT, NULL, NULL, NULL, NULL, 0, NULL);
    OCIHandleAlloc((dvoid*)envhp, (dvoid**)&errhp, OCI_HTYPE_ERROR, (size_t)0, (dvoid**)NULL);
    OCIHandleAlloc((dvoid*)envhp, (dvoid**)&srvhp, OCI_HTYPE_SERVER, (size_t)0, (dvoid**)NULL);
    OCIServerAttach(srvhp, errhp, (text*)dbname, strlen(dbname), OCI_DEFAULT);
    OCIHandleAlloc((dvoid*)envhp, (dvoid**)&svchp, OCI_HTYPE_SVCCTX, (size_t)0, (dvoid**)NULL);
    OCIHandleAlloc((dvoid*)envhp, (dvoid**)&usrhp, OCI_HTYPE_SESSION, (size_t)0, (dvoid**)NULL);
    OCILogon2(envhp, errhp, &svchp, (OraText*)username, (ub4)strlen(username), (OraText*)password, (ub4)strlen(password), (OraText*)dbname, (ub4)strlen(dbname), OCI_DEFAULT);
}

void disconnect() {
    if (svchp && errhp) OCILogoff(svchp, errhp);

    if (stmthp) OCIHandleFree((dvoid*)stmthp, OCI_HTYPE_STMT);
    if (param) OCIHandleFree((dvoid*)param, OCI_DTYPE_PARAM);
    if (usrhp) OCIHandleFree((dvoid*)usrhp, OCI_HTYPE_SESSION);
    if (svchp) OCIHandleFree((dvoid*)svchp, OCI_HTYPE_SVCCTX);
    if (srvhp) OCIHandleFree((dvoid*)srvhp, OCI_HTYPE_SERVER);
    if (errhp) OCIHandleFree((dvoid*)errhp, OCI_HTYPE_ERROR);
    if (envhp) OCIHandleFree((dvoid*)envhp, OCI_HTYPE_ENV);
}
```

```c
// transaction.c

#pragma warning(disable: 4996)
#include <stdio.h>
#include <oci.h>
#include "connection.h"
#include "transaction.h"

void check_error() {
    text errbuf[512];
    sb4 errcode = 0;
    OCIErrorGet((dvoid*)errhp, (ub4)1, (text*)NULL, &errcode, errbuf, (ub4)sizeof(errbuf), OCI_HTYPE_ERROR);
    printf("Error: %s\n", errbuf);
}

void create() {
    char* insert_sql = "INSERT INTO transactions (stock_name, trade_type, quantity, price, customer_name) VALUES (:1, :2, :3, :4, :5)";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)insert_sql, strlen(insert_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    Customer customer;
    printf("고객 이름: ");
    scanf(" %s", customer.customer_name);

    Transaction transaction;
    printf("주식 종목명: ");
    scanf(" %s", transaction.stock_name);
    do {
        printf("거래 유형 (0: 매수, 1: 매도): ");
        scanf("%d", &transaction.trade_type);
        if (transaction.trade_type != 0 && transaction.trade_type != 1) printf("거래 유형을 다시 입력하시오.\n");
    } while (transaction.trade_type != 0 && transaction.trade_type != 1);
    do {
        printf("거래 수량: ");
        scanf("%d", &transaction.quantity);
        if (transaction.quantity <= 0) printf("거래 수량을 다시 입력하시오.\n");
    } while (transaction.quantity <= 0);
    do {
        printf("거래 가격: ");
        scanf("%lf", &transaction.price);
        if (transaction.price <= 0) printf("거래 가격을 다시 입력하시오.\n");
    } while (transaction.price <= 0);

    OCIBindByPos(stmthp, &bnd1, errhp, 1, transaction.stock_name, sizeof(transaction.stock_name), SQLT_STR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByPos(stmthp, &bnd2, errhp, 2, &transaction.trade_type, sizeof(transaction.trade_type), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByPos(stmthp, &bnd3, errhp, 3, &transaction.quantity, sizeof(transaction.quantity), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByPos(stmthp, &bnd4, errhp, 4, &transaction.price, sizeof(transaction.price), SQLT_BDOUBLE, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByPos(stmthp, &bnd5, errhp, 5, customer.customer_name, sizeof(customer.customer_name), SQLT_STR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);

    if (OCIStmtExecute(svchp, stmthp, errhp, 1, 0, NULL, NULL, OCI_COMMIT_ON_SUCCESS) != OCI_SUCCESS) {
        check_error(errhp);
    }
}

void select_all() {
    char* select_sql = "SELECT transaction_id, customer_name, stock_name, trade_type, quantity, price FROM transactions";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT);

    Customer customer;
    Transaction transaction;
    OCIDefineByPos(stmthp, &def1, errhp, 1, &transaction.transaction_id, sizeof(transaction.transaction_id), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def2, errhp, 2, customer.customer_name, sizeof(customer.customer_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def3, errhp, 3, transaction.stock_name, sizeof(transaction.stock_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def4, errhp, 4, &transaction.trade_type, sizeof(transaction.trade_type), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def5, errhp, 5, &transaction.quantity, sizeof(transaction.quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def6, errhp, 6, &transaction.price, sizeof(transaction.price), SQLT_BDOUBLE, NULL, NULL, NULL, OCI_DEFAULT);

    int flag = 0;
    while ((status = OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT)) == OCI_SUCCESS || status == OCI_SUCCESS_WITH_INFO) {
        printf("ID: %d | 이름: %s | 종목명: %s | 거래유형: %s | 거래량: %d | 거래가격: %.2lf원\n",
            transaction.transaction_id,
            customer.customer_name,
            transaction.stock_name,
            transaction.trade_type ? "매도" : "매수",
            transaction.quantity,
            transaction.price
        );
        flag = 1;
    }

    if (!flag) printf("거래 내역이 존재하지 않습니다.\n");
}

void select_part() {
    char* select_sql = "SELECT transaction_id,  stock_name, trade_type, quantity, price FROM transactions where customer_name = :1";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    Customer customer;
    printf("조회할 고객 이름: ");
    scanf("%s", customer.customer_name);

    OCIBindByPos(stmthp, &bnd1, errhp, 1, customer.customer_name, sizeof(customer.customer_name), SQLT_STR, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT);

    Transaction transaction;
    OCIDefineByPos(stmthp, &def1, errhp, 1, &transaction.transaction_id, sizeof(transaction.transaction_id), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def2, errhp, 2, transaction.stock_name, sizeof(transaction.stock_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def3, errhp, 3, &transaction.trade_type, sizeof(transaction.trade_type), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def4, errhp, 4, &transaction.quantity, sizeof(transaction.quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def5, errhp, 5, &transaction.price, sizeof(transaction.price), SQLT_BDOUBLE, NULL, NULL, NULL, OCI_DEFAULT);

    int flag = 0;
    while ((status = OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT)) == OCI_SUCCESS || status == OCI_SUCCESS_WITH_INFO) {
        printf("ID: %d | 이름: %s | 종목명: %s | 거래유형: %s | 거래량: %d | 거래가격: %.2lf원\n",
            transaction.transaction_id,
            customer.customer_name,
            transaction.stock_name,
            transaction.trade_type ? "매도" : "매수",
            transaction.quantity,
            transaction.price
        );
        flag = 1;
    }

    if (!flag) printf("해당 고객의 거래 내역이 존재하지 않습니다.\n");
}

void update() {
    char* update_sql = "UPDATE transactions SET quantity = :1, price = :2 WHERE transaction_id = :3";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)update_sql, strlen(update_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    Transaction transaction;
    do {
        printf("수정할 거래 ID: ");
        scanf("%d", &transaction.transaction_id);
        if (transaction.transaction_id < 0) printf("수정할 거래 ID를 다시 입력하시오.\n");
    } while (transaction.transaction_id < 0);
    do {
        printf("수정할 거래량: ");
        scanf("%d", &transaction.quantity);
        if (transaction.quantity <= 0) printf("수정할 거래량을 다시 입력하시오.\n");
    } while (transaction.quantity <= 0);
    do {
        printf("수정할 거래가격: ");
        scanf("%lf", &transaction.price);
        if (transaction.price <= 0) printf("수정할 거래가격을 다시 입력하시오.\n");
    } while (transaction.price <= 0);

    OCIBindByPos(stmthp, &bnd1, errhp, 1, &transaction.quantity, sizeof(transaction.quantity), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByPos(stmthp, &bnd2, errhp, 2, &transaction.price, sizeof(transaction.price), SQLT_BDOUBLE, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);
    OCIBindByPos(stmthp, &bnd3, errhp, 3, &transaction.transaction_id, sizeof(transaction.transaction_id), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);

    if (OCIStmtExecute(svchp, stmthp, errhp, 1, 0, NULL, NULL, OCI_COMMIT_ON_SUCCESS) != OCI_SUCCESS) {
        check_error(errhp);
    }
}

void delete() {
    char* delete_sql = "DELETE FROM transactions WHERE transaction_id = :1";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)delete_sql, strlen(delete_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    Transaction transaction;
    do {
        printf("삭제할 거래 ID: ");
        scanf("%d", &transaction.transaction_id);
        if (transaction.transaction_id < 0) printf("삭제할 거래 ID를 다시 입력하시오.\n");
    } while (transaction.transaction_id < 0);

    OCIBindByPos(stmthp, &bnd1, errhp, 1, &transaction.transaction_id, sizeof(transaction.transaction_id), SQLT_INT, NULL, NULL, NULL, 0, NULL, OCI_DEFAULT);

    if (OCIStmtExecute(svchp, stmthp, errhp, 1, 0, NULL, NULL, OCI_COMMIT_ON_SUCCESS) != OCI_SUCCESS) {
        check_error(errhp);
    }
}

void save_file() {
    FILE* csv_file = fopen("stocks.csv", "w");
    FILE* dat_file = fopen("stocks.dat", "w");
    if (!csv_file || !dat_file) return;

    fprintf(csv_file, "ID,이름,종목명,거래유형,거래량,거래가격\n");
    fprintf(dat_file, "ID,이름,종목명,거래유형,거래량,거래가격\n");

    char* select_sql = "SELECT transaction_id, customer_name, stock_name, trade_type, quantity, price FROM transactions";
    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);

    OCIStmtExecute(svchp, stmthp, errhp, 0, 0, NULL, NULL, OCI_DEFAULT);

    Customer customer;
    Transaction transaction;
    OCIDefineByPos(stmthp, &def1, errhp, 1, &transaction.transaction_id, sizeof(transaction.transaction_id), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def2, errhp, 2, customer.customer_name, sizeof(customer.customer_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def3, errhp, 3, transaction.stock_name, sizeof(transaction.stock_name), SQLT_STR, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def4, errhp, 4, &transaction.trade_type, sizeof(transaction.trade_type), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def5, errhp, 5, &transaction.quantity, sizeof(transaction.quantity), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def6, errhp, 6, &transaction.price, sizeof(transaction.price), SQLT_BDOUBLE, NULL, NULL, NULL, OCI_DEFAULT);

    while ((status = OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT)) == OCI_SUCCESS || status == OCI_SUCCESS_WITH_INFO) {
        fprintf(csv_file, "%d,%s,%s,%s,%d,%.2lf\n",
            transaction.transaction_id,
            customer.customer_name,
            transaction.stock_name,
            transaction.trade_type ? "매도" : "매수",
            transaction.quantity,
            transaction.price
        );
        fprintf(dat_file, "%d,%s,%s,%s,%d,%.2lf\n",
            transaction.transaction_id,
            customer.customer_name,
            transaction.stock_name,
            transaction.trade_type == 1 ? "매도" : "매수",
            transaction.quantity,
            transaction.price
        );
    }
    printf("거래 내역이 저장되었습니다.\n");
}
```
