    @FXML private TableColumn<Book, String> titleCol;
    @FXML private TableColumn<Book, Integer> priceCol;

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

        searchBtn.setOnAction(e -> {
            ObservableList<Book> books = FXCollections.observableArrayList();

            books.add(new Book("123", "Java 30일 완성", 2000, "홍길동"));
            books.add(new Book("456", "Java 여기까지인가", 7000, "신사임당"));

            tableView.setItems(books);
        });
    }
}

// JavaFXJDBCTableViewController.java

import booksearch2.vo.Book;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.Button;
import javafx.scene.control.TableColumn;
import javafx.scene.control.TableView;
import javafx.scene.control.TextField;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.control.cell.TextFieldTableCell;
import javafx.util.converter.IntegerStringConverter;

import java.net.URL;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.util.ResourceBundle;

public class JavaFXJDBCTableViewController implements Initializable {
    @FXML private TableView<Book> tableView;
    @FXML private TableColumn<Book, String> isbnCol;
    @FXML private TableColumn<Book, String> titleCol;
    @FXML private TableColumn<Book, Integer> priceCol;
    @FXML private TableColumn<Book, String> authorCol;
    @FXML private TextField textField;
    @FXML private Button searchBtn;
    @FXML private Button deleteBtn;

    Connection con = null;
    PreparedStatement pstmt;
    ResultSet rs;

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        // JDBC Driver 로드
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");

            String jdbcUrl = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##DEV";
            String pw = "1234";
            con = DriverManager.getConnection(jdbcUrl, id, pw);
        } catch (Exception e) {
            e.printStackTrace();
        }

        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

        // tableView 편집 가능
        tableView.setEditable(true);
        titleCol.setCellFactory(TextFieldTableCell.forTableColumn());
        priceCol.setCellFactory(TextFieldTableCell.forTableColumn(new IntegerStringConverter()));
        authorCol.setCellFactory(TextFieldTableCell.forTableColumn());

        // 초기 화면
        searchBooks("");

        // 검색
        searchBtn.setOnAction(e -> searchBooks(textField.getText()));

        // 삭제
        deleteBtn.setOnAction(e -> deleteBook());

        // 수정
        titleCol.setOnEditCommit(e -> updateBook(e.getRowValue().getBisbn(), "btitle", e.getNewValue()));
        priceCol.setOnEditCommit(e -> updateBook(e.getRowValue().getBisbn(), "bprice", e.getNewValue().toString()));
        authorCol.setOnEditCommit(e -> updateBook(e.getRowValue().getBisbn(), "bauthor", e.getNewValue()));
    }

    public void searchBooks(String keyword) {
        String sql = "SELECT bisbn, btitle, bprice, bauthor ";
        sql += "FROM book ";
        sql += "WHERE btitle LIKE ?";

        ObservableList<Book> books = FXCollections.observableArrayList();
        try {
            // DB에서 데이터 검색
            pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "%" + keyword + "%");

            rs = pstmt.executeQuery();
            while (rs.next()) {
                Book searchBook = new Book(
                        rs.getString("bisbn"),
                        rs.getString("btitle"),
                        rs.getInt("bprice"),
                        rs.getString("bauthor")
                );
                books.add(searchBook);
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
        tableView.setItems(books);
    }

    public void deleteBook() {
        Book selectedBook = tableView.getSelectionModel().getSelectedItem();
        if (selectedBook != null) {
            String sql = "DELETE FROM book ";
            sql += "WHERE bisbn = ?";
            try {
                // DB에서 데이터 삭제
                pstmt = con.prepareStatement(sql);
                pstmt.setString(1, selectedBook.getBisbn());
                pstmt.executeUpdate();

                // tableView에서 데이터 삭제
                tableView.getItems().remove(selectedBook);
            } catch (Exception e) {
                e.printStackTrace();
            }
        }
    }

    public void updateBook(String bisbn, String column, String newValue) {
        String sql = "UPDATE book ";
        sql += "SET " + column + " = ? ";
        sql += "WHERE bisbn = ?";
        try {
            // DB에서 데이터 수정
            pstmt = con.prepareStatement(sql);
            if (column.equals("bprice")) {
                pstmt.setInt(1, Integer.parseInt(newValue));
            } else {
                pstmt.setString(1, newValue);
            }
            pstmt.setString(2, bisbn);
            pstmt.executeUpdate();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
