### 문제 1번
```java
// 책 제목에 대한 특정 keyword를 입력 받아서
// 해당 keyword가 들어있는 책을 검색하여
// 출력하는 프로그램을 작성해보시오.
// 일반 Java application으로 작성합니다.
// 책 ISBN, 책제목, 책저자, 책가격 순으로 출력합니다.

package lecture.jdbc;

import java.sql.*;
import java.util.Scanner;

public class BookSearchCommand {
    public static void main(String[] args) {
        // 리소스
        Connection con = null;
        PreparedStatement pstmt = null;
        ResultSet rs = null;

        try {
            // 1. JDBC 드라이버 등록 (최초 1회만 수행)
            Class.forName("oracle.jdbc.driver.OracleDriver");
            System.out.println("Oracle Driver 로딩을 성공했습니다.");

            // 2. 데이터베이스 연결
            //    연결 성공의 객체는 java.sql.Connection 클래스이다.
            String jdbcUrl = "jdbc:oracle:thin:@localhost:1521:xe";
            con = DriverManager.getConnection(jdbcUrl, "C##DEV", "1234");
            System.out.println("데이터베이스 연결에 성공했습니다.\n");

            Scanner sc = new Scanner(System.in);
            System.out.print("검색 Keyword를 입력하세요: ");
            String keyword = sc.nextLine();

            // 3. SQL 작성
            //    Connection 객체로부터 Statement 객체를 생성한다.
            //    (1) 일반 Statement
            //    (2) PreparedStatement
            //    (3) CallableStatement
            //    PreparedStatement의 In Parameter를 통해 SQL Injection을 예방할 수 있다.
            String sql = "SELECT bisbn, btitle, bauthor, bprice ";
            sql += "FROM book ";
            sql += "WHERE btitle LIKE ?";

            pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "%" + keyword + "%");  // parameterIndex: ?의 순서, x: 바인딩 값

            // 4. Query 실행
            //    (1) executeUpdate()
            //        SQL 구문이 INSERT, UPDATE, DELETE 일 때 사용한다.
            //        SQL 구문에 의해 영향을 받는 행의 수를 int 형태로 반환한다.
            //    (2) executeQuery()
            //        SQL 구문이 SELECT 일 때 사용한다.
            //        ResultSet을 반환하며 첫 행을 가리킨다.
            rs = pstmt.executeQuery();

            // 5. 결과처리
            //    rs.next()을 통해 다음 행으로 넘어갈 수 있는지에 대한 여부를 판단한다.
            //    rs.getInt() 또는 rs.getString()을 통해 지정된 컬럼의 값을 가져온다.
            boolean resultFlag = false;
            while (rs.next()) {
                String result = "";
                result += rs.getString("bisbn") + " ";
                result += rs.getString("btitle") + " ";
                result += rs.getString("bauthor") + " ";
                result += rs.getString("bprice") + " ";
                resultFlag = true;

                System.out.println(result);
            }

            if (!resultFlag) {
                System.out.println("조회 결과가 없습니다.");
            }

            // 6. 리소스 해제
            //    ResultSet, PreparedStatement, Connection 순으로 리소스를 해제한다.
            rs.close();
            pstmt.close();
            con .close();
            System.out.println("\n연결이 종료되었습니다.");
        } catch (ClassNotFoundException e1) {
            System.out.println("Driver Class를 찾을 수 없습니다.");
        } catch (SQLException e2) {
            System.out.println("데이터베이스 연결에 문제가 발생했습니다.");
        }
    }
}
```

### 문제 2번
```java
package lecture.jdbc;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class TransactionTest {
    public static void main(String[] args) {
        // 리소스
        Connection con = null;
        PreparedStatement pstmt;
        try {
            // 1. JDBC 드라이버 등록
            Class.forName("oracle.jdbc.driver.OracleDriver");

            // 2. 데이터베이스 연결
            String jdbcUrl = "jdbc:oracle:thin:@localhost:1521:xe";
            con = DriverManager.getConnection(jdbcUrl, "C##DEV", "1234");

            // Connection을 생성하면 기본적으로 autoCommit Mode로 설정된다.
            // 하나의 SQL 구문은 곧바로 Commit된다.
            // 이를 해제하여 Transaction으로 처리하려면 setAutoCommit() 메서드를 false로 설정한다.
            con.setAutoCommit(false);  // Transaction 시작

            // 3. SQL 작성
            String sql = "INSERT INTO memberTbl VALUES (?, ?)";

            pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "1");
            pstmt.setString(2, "홍길동");

            // 4. Query 실행
            int result = pstmt.executeUpdate();

            // 5. 결과처리
            if (result == 1) {
                System.out.println("SQL 구문이 성공적으로 실행되었습니다.");
            }

            // 명시적으로 commit() 또는 rollback()이 호출되면 Transaction이 종료된다.
            // 해당 구문이 나오지 않고 연결이 정상 종료되면, commit()으로 간주한다.
            con.commit();       // Transaction 반영 및 종료
            // con.rollback();  // Transaction 무효 및 종료

            // 6. 리소스 해제
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
