```java
// DAO.BookDAO.java

import booksearchmvc.vo.BookVO;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class BookDAO {
    public BookDAO() {
    }

    public ObservableList<BookVO> select(String keyword) {
        ObservableList<BookVO> books = FXCollections.observableArrayList();

        // Database 처리 6단계
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");

            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##DEV";
            String pw = "1234";
            Connection con = DriverManager.getConnection(url, id, pw);

            String sql = "SELECT bisbn, btitle, bprice, bauthor ";
            sql += "FROM book ";
            sql += "WHERE btitle LIKE ?";

            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "%" + keyword + "%");

            ResultSet rs = pstmt.executeQuery();
            while (rs.next()) {
                books.add(new BookVO(
                        rs.getString("bisbn"),
                        rs.getString("btitle"),
                        rs.getInt("bprice"),
                        rs.getString("bauthor")
                ));
            }

            rs.close();
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
        return books;
    }
}
```

### 3단계
사용자가 본인의 환경에 맞게 `getConnection()`을 설정하기 위해서 해당 메서드를 추상 메서드와 클래스로 분리한다. 추상 클래스로 선언한 뒤 이를 상속하는 `KUserDAO.java` 파일에서 이를 Overriding 한다.

이처럼 상위 클래스에서 기본적인 로직의 흐름을 구현하고 하위 클래스에서 구체적인 구현 내용을 변경해서 구현하는 방식을 Template Method 패턴이라고 한다.

### 6단계
`UserDAO.java`에서 여전히 `KConnectionMaker()`을 사용함으로써 Tightly Coupled 문제를 해소하지 못했다.

이에 Dependency Injection을 통해 해결한다. `UserDAO.java`의 생성자 주입을 통해 클래스 내에서 객체를 직접 생성하지 않는다.

```java
// step6.UserDAO.java

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

public class UserDAO {
    private SimpleConnectionMaker simpleConnectionMaker;

    public UserDAO(SimpleConnectionMaker simpleConnectionMaker) {
        // DI(Dependency Injection) : 생성자 주입
        // new 키워들르 통해 직접 객체를 생성하는 것이 아니라
        // 객체를 Injection 받아서 사용한다.
        this.simpleConnectionMaker = simpleConnectionMaker;
    }

    // 새로운 User 등록
    public void insert(UserVO user) {
        // 데이터베이스 처리 6단계
        try {
            Connection con = simpleConnectionMaker.makeNewConnection();

            String sql = "INSERT INTO USERS ";
            sql += "VALUES (?, ?, ?)";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, user.getId());
            pstmt.setString(2, user.getName());
            pstmt.setString(3, user.getPassword());

            pstmt.executeUpdate();

            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // id를 통해 User 검색
    public UserVO select(String userId) {
        UserVO user = null;
        // 데이터베이스 처리 6단계
        try {
            Connection con = simpleConnectionMaker.makeNewConnection();

            String sql = "SELECT * ";
            sql += "FROM USERS ";
            sql += "WHERE id = ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, userId);

            ResultSet rs = pstmt.executeQuery();
            rs.next();
            user = new UserVO(
                    rs.getString("id"),
                    rs.getString("name"),
                    rs.getString("password")
            );

            rs.close();
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
        return user;
    }
}
```
